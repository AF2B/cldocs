== BORDEAUX-THREADS:CONDITION-WAIT @ bordeaux-threads
(let* ((lock (make-lock))
       (cv (make-condition-variable))
       (ready nil)
       (consumer (make-thread
                  (lambda ()
                    (with-lock-held (lock)
                      (loop until ready do (condition-wait cv lock))
                      :got-signal)))))
  (with-lock-held (lock)
    (setf ready t)
    (condition-notify cv))
  (join-thread consumer))
