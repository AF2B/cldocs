== BORDEAUX-THREADS:WITH-LOCK-HELD @ bordeaux-threads
(let* ((lock (make-lock "counter"))
       (counter 0)
       (threads (loop repeat 4
                      collect (make-thread
                               (lambda ()
                                 (dotimes (i 100)
                                   (with-lock-held (lock)
                                     (incf counter))))))))
  (mapc #'join-thread threads)
  counter)
