== LOCAL-TIME:TIMEZONES-MATCHING-SUBZONE @ local-time
-- silent
(setf *default-timezone* +utc-zone+)
(progn (reread-timezone-repository)
       (let ((matches (timezones-matching-subzone "EST" (parse-timestring "2024-01-15T12:00:00Z"))))
         (not (null (find "America/New_York" matches
                          :key (lambda (match) (zone-name (first match)))
                          :test #'string=)))))
