  (:merge :standard)

                     (declare (ignore char))
                     (list 'quote (read stream t nil t)))))

  (values (read-from-string "?(a b c)")))

== EDITOR-HINTS.NAMED-READTABLES:DEFREADTABLE @ named-readtables

(defun read-bracket-list (stream char)
  (declare (ignore char))
  (cons 'list (read-delimited-list #\] stream t)))
-- silent
(defreadtable bracket-syntax
  (:merge :standard)
  (:macro-char #\[ #'read-bracket-list)
  (:macro-char #\] (get-macro-character #\))))
(let ((*readtable* (ensure-readtable 'bracket-syntax)))
  (eval (read-from-string "[1 2 (+ 1 2)]")))

== EDITOR-HINTS.NAMED-READTABLES:MERGE-READTABLES-INTO @ named-readtables
-- silent
(defreadtable quote-syntax
  (:merge :standard)
  (:macro-char #\! (lambda (stream char)
                     (declare (ignore char))
                     (list 'quote (read stream t nil t)))))
(let ((target (make-readtable)))
  (merge-readtables-into target :standard 'quote-syntax)
  (let ((*readtable* target))
    (values (read-from-string "!(x y)"))))

== EDITOR-HINTS.NAMED-READTABLES:READER-MACRO-CONFLICT @ named-readtables

  (:merge :standard)

  (reader-macro-conflict () :conflict))
